CREATE DATABASE IF NOT EXISTS essence
    CHARACTER SET utf8mb4
    COLLATE utf8mb4_unicode_ci;

USE essence;

-- Cria a tabela caso ela ainda não exista
CREATE TABLE IF NOT EXISTS usuarios (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    email VARCHAR(150) NOT NULL UNIQUE,
    senha VARCHAR(255) NOT NULL
);

-- Adiciona as colunas novas do cadastro completo
-- (ficam vazias para usuários que já estavam cadastrados)
ALTER TABLE usuarios
    ADD COLUMN IF NOT EXISTS cpf         VARCHAR(11)  NULL,
    ADD COLUMN IF NOT EXISTS telefone    VARCHAR(11)  NULL,
    ADD COLUMN IF NOT EXISTS cep         VARCHAR(8)   NULL,
    ADD COLUMN IF NOT EXISTS rua         VARCHAR(150) NULL,
    ADD COLUMN IF NOT EXISTS numero      VARCHAR(10)  NULL,
    ADD COLUMN IF NOT EXISTS bairro      VARCHAR(100) NULL,
    ADD COLUMN IF NOT EXISTS cidade      VARCHAR(100) NULL,
    ADD COLUMN IF NOT EXISTS complemento VARCHAR(100) NULL,
    ADD COLUMN IF NOT EXISTS referencia  VARCHAR(150) NULL,
    ADD COLUMN IF NOT EXISTS criado_em   TIMESTAMP DEFAULT CURRENT_TIMESTAMP;

-- Impede dois cadastros com o mesmo CPF
ALTER TABLE usuarios
    ADD UNIQUE INDEX IF NOT EXISTS uk_usuarios_cpf (cpf);

        function renderizarProdutos(categoria = 'todos') {
            listaProdutos.innerHTML = '';

                card.dataset.id = produto.id;
                card.dataset.nome = produto.nome;
                card.dataset.preco = produto.preco;

                card.innerHTML = `
                    <img src="${produto.imagem}" alt="${produto.nome}">
                    <div class="product-info">
                        <h3>${produto.nome}</h3>
                        <p class="descricao">${produto.descricao || ''}</p>
                        <p class="price">${formatarPreco(produto.preco)}</p>
                        <button class="btn btn-comprar">Comprar</button>
                        <button class="btn btn-adicionar">Adicionar</button>
                    </div>
                `;

                listaProdutos.appendChild(card);
            });

            ativarBotoes();
        }

        // ==========================================================
        // CARRINHO (mesmo padrão usado nas outras páginas do site)
        // ==========================================================
        function ativarBotoes() {
            const cards = document.querySelectorAll('.product-card');

            cards.forEach(card => {
                const btnComprar = card.querySelector('.btn-comprar');
                const btnAdicionar = card.querySelector('.btn-adicionar');

                const produto = {
                    id: card.dataset.id,
                    nome: card.dataset.nome,
                    preco: parseFloat(card.dataset.preco)
                };

                btnComprar.addEventListener('click', () => {
                    exigirLogin(() => {
                        localStorage.setItem('compraDireta', JSON.stringify(produto));
                        window.location.href = 'checkout.html';
                    });
                });

                btnAdicionar.addEventListener('click', () => {
                    exigirLogin(() => {
                        adicionarAoCarrinho(produto);
                        alert(`${produto.nome} adicionado ao carrinho!`);
                    });
                });
            });
        }

        function adicionarAoCarrinho(produto) {
            let carrinho = JSON.parse(localStorage.getItem('carrinho')) || [];

            const existente = carrinho.find(item => item.id === produto.id);
            if (existente) {
                existente.quantidade += 1;
            } else {
                carrinho.push({ ...produto, quantidade: 1 });
            }

            localStorage.setItem('carrinho', JSON.stringify(carrinho));
        }

        // ==========================================================
        // CARRINHO — lê os produtos salvos em localStorage('carrinho')
        // pelas páginas de produtos (mesmo padrão do site)
        // ==========================================================
        const conteudo = document.getElementById('carrinhoConteudo');

        const formatarPreco = valor => `R$ ${valor.toFixed(2).replace('.', ',')}`;

        function obterCarrinho(){
            return JSON.parse(localStorage.getItem('carrinho')) || [];
        }

        function salvarCarrinho(carrinho){
            localStorage.setItem('carrinho', JSON.stringify(carrinho));

        function renderizarCarrinho(){
            const carrinho = obterCarrinho();

            if(carrinho.length === 0){
                conteudo.innerHTML = `
                    <div class="carrinho-vazio">
                        <p>Seu carrinho está vazio.</p>
                        <a href="tcc.html" class="btn">Continuar comprando</a>
                    </div>
                `;
                return;
            }

            const totalItens = carrinho.reduce((soma, item) => soma + item.quantidade, 0);
            const subtotal = carrinho.reduce((soma, item) => soma + (item.preco * item.quantidade), 0);

            const itensHtml = carrinho.map(item => `
                <div class="item-carrinho" data-id="${item.id}">
                    ${item.imagem ? `<img src="${item.imagem}" alt="${item.nome}">` : `<img src="" alt="${item.nome}">`}
                    <div class="item-info">
                        <h3>${item.nome}</h3>
                        <p class="item-preco-unit">${formatarPreco(item.preco)} / unidade</p>
                        <div class="item-qtd">
                            <button class="qtd-btn" data-acao="diminuir" data-id="${item.id}">−</button>
                            <span class="qtd-valor">${item.quantidade}</span>
                            <button class="qtd-btn" data-acao="aumentar" data-id="${item.id}">+</button>
                        </div>
                    </div>
                    <div class="item-subtotal">${formatarPreco(item.preco * item.quantidade)}</div>
                    <button class="remover-btn" data-acao="remover" data-id="${item.id}" title="Remover item">&times;</button>
                </div>
            `).join('');

            conteudo.innerHTML = `
                <div class="carrinho-wrap">
                    <div class="lista-itens">
                        ${itensHtml}
                    </div>
                    <div class="resumo">
                        <h2>Resumo do pedido</h2>
                        <div class="resumo-linha">
                            <span>Itens (${totalItens})</span>
                            <span>${formatarPreco(subtotal)}</span>
                        </div>
                        <div class="resumo-linha">
                            <span>Frete</span>
                            <span>A calcular</span>
                        </div>
                        <div class="resumo-total">
                            <span>Total</span>
                            <span>${formatarPreco(subtotal)}</span>
                        </div>
                        <button class="btn" id="btnFinalizar">Finalizar compra</button>
                        <a href="tcc.html" class="continuar-link">Continuar comprando</a>
                    </div>
                </div>
            `;

            document.getElementById('btnFinalizar').addEventListener('click', () => {
                window.location.href = 'checkout.html';
            });
        }

        conteudo.addEventListener('click', e => {
            const btn = e.target.closest('[data-acao]');
            if(!btn) return;

            const id = btn.dataset.id;
            const acao = btn.dataset.acao;
            let carrinho = obterCarrinho();
            const item = carrinho.find(p => p.id === id);
            if(!item) return;

            if(acao === 'aumentar'){
                item.quantidade += 1;
            } else if(acao === 'diminuir'){
                item.quantidade -= 1;
                if(item.quantidade <= 0){
                    carrinho = carrinho.filter(p => p.id !== id);
                }
            } else if(acao === 'remover'){
                carrinho = carrinho.filter(p => p.id !== id);
            }

            salvarCarrinho(carrinho);
            renderizarCarrinho();
        });

        renderizarCarrinho();
    </script>

        const formatarPreco = valor => `R$ ${valor.toFixed(2).replace('.', ',')}`;

        const listaProdutos = document.getElementById('listaProdutos');

        function renderizarProdutos(categoria = 'todos') {
            listaProdutos.innerHTML = '';

            const produtosFiltrados = categoria === 'todos'
                ? produtosCorpo
                : produtosCorpo.filter(p => p.categoria === categoria);

            produtosFiltrados.forEach(produto => {
                const card = document.createElement('div');
                card.className = 'product-card';
                card.dataset.id = produto.id;
                card.dataset.nome = produto.nome;
                card.dataset.preco = produto.preco;

                card.innerHTML = `
                    <img src="${produto.imagem}" alt="${produto.nome}">
                    <div class="product-info">
                        <h3>${produto.nome}</h3>
                        <p class="descricao">${produto.descricao || ''}</p>
                        <p class="price">${formatarPreco(produto.preco)}</p>
                        <button class="btn btn-comprar">Comprar</button>
                        <button class="btn btn-adicionar">Adicionar</button>
                    </div>
                `;

                listaProdutos.appendChild(card);
            });

            ativarBotoes();
        }

        renderizarProdutos();

        // ---------- FILTRO DE CATEGORIAS ----------
        document.getElementById('filtros').addEventListener('click', e => {
            const btn = e.target.closest('.filtro-btn');
            if (!btn) return;
            document.querySelectorAll('.filtro-btn').forEach(b => b.classList.remove('ativo'));
            btn.classList.add('ativo');
            renderizarProdutos(btn.dataset.cat);
        });

        // ==========================================================
        // CARRINHO (sem depender de exigirLogin/auth.js)
        // ==========================================================
        function ativarBotoes() {
            const cards = document.querySelectorAll('.product-card');

            cards.forEach(card => {
                const btnComprar = card.querySelector('.btn-comprar');
                const btnAdicionar = card.querySelector('.btn-adicionar');

                const produto = {
                    id: card.dataset.id,
                    nome: card.dataset.nome,
                    preco: parseFloat(card.dataset.preco)
                };

                btnComprar.addEventListener('click', () => {
                    localStorage.setItem('compraDireta', JSON.stringify(produto));
                    window.location.href = 'checkout.html';
                });

                btnAdicionar.addEventListener('click', () => {
                    adicionarAoCarrinho(produto);
                    alert(`${produto.nome} adicionado ao carrinho!`);
                });
            });
        }

        function adicionarAoCarrinho(produto) {
            let carrinho = JSON.parse(localStorage.getItem('carrinho')) || [];

            const existente = carrinho.find(item => item.id === produto.id);
            if (existente) {
                existente.quantidade += 1;
            } else {
                carrinho.push({ ...produto, quantidade: 1 });
            }

            localStorage.setItem('carrinho', JSON.stringify(carrinho));
        }
    </script>

        function adicionarAoCarrinho(produto) {
            let carrinho = JSON.parse(localStorage.getItem('carrinho')) || [];

            const existente = carrinho.find(item => item.id === produto.id);
            if (existente) {
                existente.quantidade += 1;
            } else {
                carrinho.push({ ...produto, quantidade: 1 });
            }

            localStorage.setItem('carrinho', JSON.stringify(carrinho));

        function ativarBotoes() {
            const cards = document.querySelectorAll('.product-card');

            cards.forEach(card => {
                const btnComprar = card.querySelector('.btn-comprar');
                const btnAdicionar = card.querySelector('.btn-adicionar');

                    id: card.dataset.id,
                    nome: card.dataset.nome,
                    preco: parseFloat(card.dataset.preco)
                };

                    localStorage.setItem('compraDireta', JSON.stringify(produto));
                    window.location.href = 'checkout.html';

        function adicionarAoCarrinho(produto) {
            let carrinho = JSON.parse(localStorage.getItem('carrinho')) || [];

            const existente = carrinho.find(item => item.id === produto.id);
            if (existente) {
                existente.quantidade += 1;
            } else {
                carrinho.push({ ...produto, quantidade: 1 });
            }

            localStorage.setItem('carrinho', JSON.stringify(carrinho));
        }

        function atualizarContagem() {
            const agora = new Date();
            const diff = dataFinal - agora;

            if (diff <= 0) {
                document.getElementById('contagem').innerHTML = '<p>Promoção encerrada</p>';
                return;
            }

            const dias = Math.floor(diff / (1000 * 60 * 60 * 24));
            const horas = Math.floor((diff / (1000 * 60 * 60)) % 24);
            const min = Math.floor((diff / (1000 * 60)) % 60);
            const seg = Math.floor((diff / 1000) % 60);

            document.getElementById('cd-dias').textContent = String(dias).padStart(2, '0');
            document.getElementById('cd-horas').textContent = String(horas).padStart(2, '0');
            document.getElementById('cd-min').textContent = String(min).padStart(2, '0');
            document.getElementById('cd-seg').textContent = String(seg).padStart(2, '0');
        }

        function adicionarAoCarrinho(produto) {
            let carrinho = JSON.parse(localStorage.getItem('carrinho')) || [];

            const existente = carrinho.find(item => item.id === produto.id);
            if (existente) {
                existente.quantidade += 1;
            } else {
                carrinho.push({ ...produto, quantidade: 1 });
            }

            localStorage.setItem('carrinho', JSON.stringify(carrinho));
        }

// Traduz as mensagens de erro mais comuns do Supabase
function traduzirErro(mensagem) {
    const m = (mensagem || '').toLowerCase();
    if (m.includes('invalid login credentials')) return 'E-mail ou senha incorretos.';
    if (m.includes('already registered')) return 'Este e-mail já está cadastrado.';
    if (m.includes('password should be at least')) return 'A senha precisa ter pelo menos 6 caracteres.';
    if (m.includes('email not confirmed')) return 'Confirme seu e-mail antes de entrar (veja sua caixa de entrada).';
    if (m.includes('invalid') && m.includes('email')) return 'Digite um e-mail válido.';
    if (m.includes('rate limit')) return 'Muitas tentativas. Aguarde alguns minutos e tente de novo.';
    return 'Ocorreu um erro: ' + mensagem;
}

// Troca "Cadastrar / Login" por "Olá, nome / Sair" quando a pessoa está logada
async function atualizarMenu() {
    const linkCadastrar = document.getElementById('link-cadastrar');
    const linkLogin = document.getElementById('link-login');
    if (!linkCadastrar || !linkLogin) return;

    const { data: { user } } = await db.auth.getUser();
    if (!user) return;

    const nome = (user.user_metadata && user.user_metadata.nome) || user.email.split('@')[0];
    linkCadastrar.textContent = 'Olá, ' + nome.split(' ')[0];
    linkCadastrar.href = '#';

    linkLogin.textContent = 'Sair';
    linkLogin.href = '#';
    linkLogin.addEventListener('click', async (e) => {
        e.preventDefault();
        await db.auth.signOut();
        window.location.href = 'index.html';
    });
}

    <script>
    // ==========================================================
    // CARRINHO (sem depender de exigirLogin/auth.js)
    // ==========================================================
    const cards = document.querySelectorAll('.product-card');

    cards.forEach(card => {
        const btnComprar = card.querySelector('.btn-comprar');
        const btnAdicionar = card.querySelector('.btn-adicionar');

        const produto = {
            id: card.dataset.id,
            nome: card.dataset.nome,
            preco: parseFloat(card.dataset.preco)
        };

        btnComprar.addEventListener('click', () => {
            localStorage.setItem('compraDireta', JSON.stringify(produto));
            window.location.href = 'checkout.html';
        });

        btnAdicionar.addEventListener('click', () => {
            adicionarAoCarrinho(produto);
            alert(`${produto.nome} adicionado ao carrinho!`);
        });
    });

    function adicionarAoCarrinho(produto) {
        let carrinho = JSON.parse(localStorage.getItem('carrinho')) || [];

        const existente = carrinho.find(item => item.id === produto.id);
        if (existente) {
            existente.quantidade += 1;
        } else {
            carrinho.push({ ...produto, quantidade: 1 });
        }

        localStorage.setItem('carrinho', JSON.stringify(carrinho));
    }

    // ---------- NEWSLETTER (por enquanto só visual) ----------
    const formNewsletter = document.getElementById('formNewsletter');
    if(formNewsletter){
        formNewsletter.addEventListener('submit', e => {
            e.preventDefault();
            const msg = document.getElementById('newsletterMensagem');
            msg.textContent = 'Inscrição realizada! Fique de olho no seu e-mail.';
            msg.style.display = 'block';
            formNewsletter.reset();
        });
    }
</script>
